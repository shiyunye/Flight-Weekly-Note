  select 
   scenario
  ,product_detail
  ,trans_date
  ,DATE_ADD(DATE_TRUNC(Trans_Date, WEEK(SUNDAY)), INTERVAL 6 DAY) wk_ending
  ,format_date('%Y-%m-%d', DATE_TRUNC(trans_date, MONTH)) year_month
  ,sum(gr_contr) as gr_contr
  ,sum(ref_contr) as ref_contr 
  ,sum(gr_fee) as gr_fee
  ,sum(ref_fee) as ref_fee 
  ,sum(gds_incentives) as gds_incentives
  ,sum(wex_rebates) as wex_rebates
  ,sum(gr_contr + ref_contr + gr_fee + ref_fee + gds_incentives + wex_rebates) as transactional_revenue
  ,SUM(IF(product_detail = 'AIR CORE',
        gr_contr + ref_contr + gr_fee + ref_fee + gds_incentives+wex_rebates, 0)) AS net_cont_fee_GDS_Wex
  ,SUM(IF(product_detail = 'AIR CORE',
        gr_contr + gr_fee + gds_incentives + wex_rebates, 0)) AS gr_cont_fee_GDS_Wex
  ,sum(gr_units) as gr_Units
  ,sum(gr_units + ref_units) as net_units,
  
  from `pcln-pl-spaanalytics-prod.spa_datamart.FPA_REPORTING_LIVE_V`
WHERE scenario in ('PLAN')
  AND view_desc = 'ACTIVITY'
  AND Trans_Date >= ('2025-12-01')
  and product_detail LIKE '%AIR%'
  group by 1,2,3,4,5
order by 1,2,3
